## Why T032's method no longer measures the child (verified 2026-10-19)

## Method (host step, no credentials, no network)

A watcher (kept outside the repository at `~/.local/share/ovh-lz/review/t050-006-peaks.py`, inlined below) starts the entry itself, so it knows the entry's pid, and polls every 2 ms for `user@<uid>.service/{,*/,*/*/}lz-offline-<that pid>-*.scope`. When the scope appears it starts a pin (`/usr/bin/sleep infinity`) and writes the pin's pid to the scope's `cgroup.procs` (allowed: both sit under the user-owned, delegated `app.slice`). The pin keeps the scope alive after the holder exits, so after the entry has exited the watcher reads the scope's final `memory.peak`, `pids.peak`, `memory.max`, `pids.max`, `memory.swap.max`, `memory.events` `oom_kill` and `pids.events` `max`, checks the pin is in the scope, then kills the pin and the scope is collected. It also keeps the running maximum of the polled peaks. Effects of the pin: +1 to `pids.peak` from the moment it joins (included in the figures below); memory already charged stays with the pin's old cgroup (cgroup v2 does not move charges), but `sleep` may still be starting when it is moved, so a few hundred KB of its own may be charged to the scope: if anything an overstatement of the peak, which errs towards the limits. `memory.peak` includes page cache charged to the scope, as in CI.

Run script `t050-006-run.sh` (outside the repository), from `$HOME`: the watcher over the 20 `LZ_TARGETS` of `.github/workflows/foundation.yml:21` in that order, entry r5, candidate this worktree at `ab1d417` plus T050's task line (Markdown only; the code tree is `adf671f`'s), then the outer-scope demonstration above. Swap state recorded before and after.

```python
#!/usr/bin/env python3
"""T050 (spec 006) host step: peaks of the entry's OWN child scope.

usage: t050-006-peaks.py <out-file> entry <entry> <candidate> <target> [<target> ...]
       t050-006-peaks.py <out-file> control <bytes> <forks>
"""
import glob
import os
import subprocess
import sys
import time

UID = os.getuid()
BASE = f"/sys/fs/cgroup/user.slice/user-{UID}.slice/user@{UID}.service"
CANDIDATE_GLOBS = [BASE + "/lz-offline-{pid}-*.scope", BASE + "/*/lz-offline-{pid}-*.scope",
                   BASE + "/*/*/lz-offline-{pid}-*.scope"]


def read(path):
    try:
        with open(path) as f:
            return f.read().strip()
    except OSError:
        return None


def event(text, key):
    for line in (text or "").splitlines():
        name, _, value = line.partition(" ")
        if name == key:
            return int(value)
    return None


def find_scope(pid):
    for pattern in CANDIDATE_GLOBS:
        hits = glob.glob(pattern.format(pid=pid))
        if hits:
            return hits
    return []


def watch(argv, log):
    start = time.monotonic()
    proc = subprocess.Popen(argv, stdout=log, stderr=subprocess.STDOUT, cwd=os.environ["HOME"])
    scope, pin, pin_err = None, None, None
    polled_mem, polled_pids, found_at = 0, 0, None
    while True:
        exited = proc.poll() is not None
        if scope is None:
            hits = find_scope(proc.pid)
            if len(hits) > 1:
                scope = "AMBIGUOUS:" + ",".join(hits)
            elif hits:
                scope = hits[0]
                found_at = time.monotonic() - start
                pin = subprocess.Popen(["/usr/bin/sleep", "infinity"])
                try:
                    with open(os.path.join(scope, "cgroup.procs"), "w") as f:
                        f.write(str(pin.pid))
                except OSError as err:
                    pin_err = repr(err)
        if scope and not scope.startswith("AMBIGUOUS"):
            m = read(os.path.join(scope, "memory.peak"))
            p = read(os.path.join(scope, "pids.peak"))
            if m and m.isdigit():
                polled_mem = max(polled_mem, int(m))
            if p and p.isdigit():
                polled_pids = max(polled_pids, int(p))
        if exited:
            break
        time.sleep(0.002)
    code = proc.returncode
    seconds = time.monotonic() - start
    result = {"exit": code, "seconds": round(seconds, 1), "scope": scope and os.path.basename(scope),
              "found_at": found_at and round(found_at, 3), "pin_err": pin_err,
              "polled_memory_peak": polled_mem, "polled_pids_peak": polled_pids}
    if scope and not scope.startswith("AMBIGUOUS"):
        pin_cg = read(f"/proc/{pin.pid}/cgroup") if pin else None
        result["pin_in_scope"] = bool(pin_cg and pin_cg.endswith("/" + os.path.basename(scope)))
        result["memory_peak"] = read(os.path.join(scope, "memory.peak"))
        result["pids_peak"] = read(os.path.join(scope, "pids.peak"))
        result["memory_max"] = read(os.path.join(scope, "memory.max"))
        result["pids_max"] = read(os.path.join(scope, "pids.max"))
        result["swap_max"] = read(os.path.join(scope, "memory.swap.max"))
        result["oom_kill"] = event(read(os.path.join(scope, "memory.events")), "oom_kill")
        result["pids_max_events"] = event(read(os.path.join(scope, "pids.events")), "max")
        result["swap_peak"] = read(os.path.join(scope, "memory.swap.peak"))
    if pin:
        pin.kill()
        pin.wait()
    return result


def main():
    out, mode = sys.argv[1], sys.argv[2]
    with open(out, "a") as rec:
        if mode == "control":
            nbytes, forks = int(sys.argv[3]), int(sys.argv[4])
            prog = ("import subprocess,time\n"
                    f"b=b'\\x01'*{nbytes}\n"
                    f"ps=[subprocess.Popen(['/usr/bin/sleep','3']) for _ in range({forks})]\n"
                    "[p.wait() for p in ps]\n")
            # sh's pid is the Popen pid; exec keeps it, so the unit carries the watched pid.
            argv = ["/bin/sh", "-c",
                    'exec /usr/bin/systemd-run --user --scope --quiet --collect --unit=lz-offline-$$-control '
                    '-p MemoryMax=4294967296 -p MemorySwapMax=0 -p TasksMax=1024 -- /usr/bin/python3 -I -c "$1"',
                    "control", prog]
            with open(out + ".control.log", "w") as log:
                r = watch(argv, log)
            print("LZ_T050_CONTROL", f"bytes={nbytes}", f"forks={forks}",
                  " ".join(f"{k}={v}" for k, v in r.items()), file=rec, flush=True)
            return
        entry, candidate = sys.argv[3], sys.argv[4]
        for target in sys.argv[5:]:
            logpath = out + "." + target.replace(":", "-") + ".log"
            with open(logpath, "w") as log:
                r = watch([entry, "--candidate", candidate, "--", "task", target], log)
            print("LZ_T050_PEAK", target, " ".join(f"{k}={v}" for k, v in r.items()), file=rec, flush=True)


if __name__ == "__main__":
    main()
```
